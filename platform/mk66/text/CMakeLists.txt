# platform/mk66/text (OS-FLASHDIET P5/P8): the platform's long texts as deflate blobs. text/*.txt (one string per line) is packed by tools/text_pack.py, with
# tools/deflate_pack.py, into a generated blob source and an ids header per pool in the build tree; `platform_mk66_text` holds the blobs and the pure decode
# (text_unpack), and joins platform_mk66_pure, so the host and QEMU tests read the same bytes the image serves. The image's pools and text() (src/text.cpp)
# are added to the image by the block of platform/mk66/CMakeLists.txt.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(_text_gen "${CMAKE_CURRENT_BINARY_DIR}/gen")
set(_text_ids_dir "${_text_gen}/o3os/platform/mk66")

# One pool: <txt file> -> <symbol> blob and <ns> ids.
function(o3os_text_pool stem symbol ns)
  set(src "${CMAKE_CURRENT_SOURCE_DIR}/${stem}.txt")
  set(ids "${_text_ids_dir}/${stem}_ids.h")
  set(blob "${_text_gen}/${stem}_blob.cpp")
  set(expect "${_text_ids_dir}/${stem}_expect.h")  # tests only: every text as its literal
  add_custom_command(
    OUTPUT "${ids}" "${blob}" "${expect}"
    COMMAND ${Python3_EXECUTABLE} -I "${PROJECT_SOURCE_DIR}/tools/text_pack.py"
            --text "${src}" --symbol ${symbol} --ns ${ns} --ids "${ids}" --out "${blob}" --expect "${expect}"
    DEPENDS "${src}" "${PROJECT_SOURCE_DIR}/tools/text_pack.py" "${PROJECT_SOURCE_DIR}/tools/deflate_pack.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/include/o3os/platform/mk66/text.h"
    COMMENT "Packing ${stem} (tools/text_pack.py)"
    VERBATIM)
  set(${stem}_BLOB "${blob}" PARENT_SCOPE)
  set(${stem}_IDS "${ids}" PARENT_SCOPE)
endfunction()

o3os_text_pool(console_text ConsoleText txt)
o3os_text_pool(bench_text BenchText btxt)

add_library(platform_mk66_text STATIC src/text_unpack.cpp "${console_text_BLOB}" "${bench_text_BLOB}")
set_target_properties(platform_mk66_text PROPERTIES OUTPUT_NAME o3os_platform_mk66_text)
target_include_directories(platform_mk66_text PUBLIC include "${_text_gen}")
target_link_libraries(platform_mk66_text PUBLIC core)
o3os_strict_warnings(platform_mk66_text)
# The ids headers are generated by the same commands as the blobs: whoever includes one builds after them.
add_custom_target(platform_mk66_text_ids DEPENDS "${console_text_IDS}" "${bench_text_IDS}" "${_text_ids_dir}/console_text_expect.h" "${_text_ids_dir}/bench_text_expect.h")
add_dependencies(platform_mk66_text platform_mk66_text_ids)
target_link_libraries(platform_mk66_pure PUBLIC platform_mk66_text)
add_dependencies(platform_mk66_pure platform_mk66_text_ids)
