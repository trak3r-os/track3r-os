# Tests registered here (this directory is added before the root's enable_testing(), so it asks for it itself):
#   scenario_<name>           runs scenarios/<name>.txt twice (determinism: the framebuffers, and the samples of every `render`) and compares
#                             every framebuffer's SHA-256 with goldens.txt; the renders go to <build>/sim/wav/ as WAV files (OS-E2E-2)
#   gallery_<i>of<N>          `o3sim --gallery --shard i/N` renders every N-th shot of the registry (screens.cpp, all
#                             states x themes) twice, prints the display-list budget table (a breach is exit 1), writes
#                             the PNGs, the contact sheet and diff_list.txt, and compares the hashes with goldens.txt;
#                             the N shards together are the whole gallery (a run took 200 s in one piece under the sanitizers)
#   gallery_shards_cover      the shards' key lists (`--list`, nothing drawn) add up to the whole gallery, each shot once
#   o3sim_*_exit<code>        o3sim's exit codes: 2 for an invalid scenario and for frames that do not fit a
#                             shrunken display list, 1 for a golden that does not match (negative/)
# The PNG files are checked against the hashed pixels by the unit tests (sha256_png_test.cpp).
enable_testing()

# OS-E2E-4: the audio fingerprints of the journeys that opt in with a `fingerprint` line (tests/audio_fingerprints/<scenario>.json, sim/include/o3os/sim/audio_fingerprint.h). Every scenario test compares its
# own with the stored one (a scenario with no `fingerprint` line has no file: a stale file is a failure), in the same process as the run, so a fingerprint costs a journey nothing more.
set(_fingerprints ${PROJECT_SOURCE_DIR}/tests/audio_fingerprints)
set(_scenario_names "")
foreach(_s IN LISTS _scenarios)
  get_filename_component(_n ${_s} NAME_WE)
  if(_n IN_LIST _scenario_names)
    message(FATAL_ERROR "two scenarios are called '${_n}' (${_s}): a name is the ctest name and the goldens prefix; give the file its folder's prefix (qa_, e2e_)")
  endif()
  list(APPEND _scenario_names ${_n})
  set(_png ${CMAKE_CURRENT_BINARY_DIR}/png/${_n})
  add_test(NAME scenario_${_n}
    COMMAND o3sim --scenario ${_s} --out ${_png} --goldens ${_goldens} --check --twice --cards ${_cards} --wav ${CMAKE_CURRENT_BINARY_DIR}/wav
            --fingerprints ${_fingerprints} --check-fingerprints)
  set_tests_properties(scenario_${_n} PROPERTIES TIMEOUT 120)
  # OS-QANET D1: a journey of e2e/ that uses a card of the list above runs again on the FAT32 image of it, with the same asserts, goldens and fingerprints (a scenario that needs a folder of the card, `apps on`, is
  # not one of them). The picture and WAV folders are its own.
  get_filename_component(_dir ${_s} DIRECTORY)
  get_filename_component(_folder ${_dir} NAME)
  if(_folder STREQUAL "e2e")
    file(STRINGS ${_s} _card_lines REGEX "^card (e2e_|qanet_card)")
    if(_card_lines)
      add_test(NAME scenario_fatfs_${_n}
        COMMAND o3sim --scenario ${_s} --out ${CMAKE_CURRENT_BINARY_DIR}/png_fat/${_n} --goldens ${_goldens} --check --twice --cards ${_cards_fat} --wav ${CMAKE_CURRENT_BINARY_DIR}/wav_fat
                --fingerprints ${_fingerprints} --check-fingerprints)
      set_tests_properties(scenario_fatfs_${_n} PROPERTIES TIMEOUT 120)
    endif()
  endif()
endforeach()

function(o3sim_exit_test name code stderr_regex)  # remaining arguments: o3sim's own
  string(REPLACE ";" "|" _args "$<TARGET_FILE:o3sim>;${ARGN}")
  add_test(NAME ${name}
    COMMAND ${CMAKE_COMMAND} -DEXPECT=${code} "-DSTDERR=${stderr_regex}" "-DCOMMAND=${_args}"
            -P ${CMAKE_CURRENT_SOURCE_DIR}/expect_exit.cmake)
endfunction()

# OS-E2E-9: the reference players inside journeys (docs/specs/OS-E2E.md "E2E-9"). Each runs a journey of scenarios/e2e/ in the simulator, reads the captured render back and compares it with the
# reference's own render of the same bytes. Label `e2e` (cmake/O3osLabels.cmake): tools/ci.sh e2e runs them, `ci.sh host` leaves them out. A missing library (libopenmpt, numpy, scipy) is a skip
# (exit 77), and a failure under O3_CI=1, which tools/ci.sh sets. The `_break_*` tests perturb the REFERENCE on purpose and must report disagreements: the comparison can fail. 19 of them:
# a `_break_<kind>` moves its reference a long way (10 to 50 times the tolerance: the comparison fails at all), a `_break_near_<kind>` by about TWICE the tolerance (it fails close to it), and
# `e2e_ref_dsp_limiter_break_clipper` replaces the limiter's two-voice take by what a hard clipper makes of it (the static curve passes it; the gain and the distortion checks must not).
# A break is `<script argument>:<name>:<--perturb argument>`; the regular expression makes sure that the report names disagreements, so a crash does not pass.
set(_e2e_break_regex "comparisons, [1-9][0-9]* disagree")
set(_e2e_ref_dir ${CMAKE_CURRENT_BINARY_DIR}/e2e_ref)
set(_openmpt_ref ${CMAKE_CURRENT_SOURCE_DIR}/../tests/e2e_openmpt_ref.py)

  add_test(NAME e2e_ref_openmpt_${_f}
    COMMAND ${Python3_EXECUTABLE} -I ${_openmpt_ref} ${_f} $<TARGET_FILE:o3sim> ${_cards} ${CMAKE_CURRENT_SOURCE_DIR}/scenarios/e2e ${_e2e_ref_dir}/openmpt_${_f})
  set_tests_properties(e2e_ref_openmpt_${_f} PROPERTIES TIMEOUT 300 SKIP_RETURN_CODE 77)
endforeach()
foreach(_b mod:pitch:pitch s3m:tempo:tempo med:level:level paula:amiga:amiga mod:near_pitch:pitch=3 med:near_level:level=0.9 paula:near_tilt:tilt=3)
  string(REPLACE ":" ";" _bp ${_b})
  list(GET _bp 0 _f)
  list(GET _bp 1 _kind)
  list(GET _bp 2 _arg)
  add_test(NAME e2e_ref_openmpt_${_f}_break_${_kind}
    COMMAND ${Python3_EXECUTABLE} -I ${_openmpt_ref} ${_f} $<TARGET_FILE:o3sim> ${_cards} ${CMAKE_CURRENT_SOURCE_DIR}/scenarios/e2e ${_e2e_ref_dir}/openmpt_${_f}_break_${_kind} --perturb ${_arg})
  set_tests_properties(e2e_ref_openmpt_${_f}_break_${_kind} PROPERTIES TIMEOUT 300 SKIP_RETURN_CODE 77 PASS_REGULAR_EXPRESSION "${_e2e_break_regex}")
endforeach()
