# OS-FINALQA (docs/specs/OS-FINALQA.md): the Config rows of docs/qa/catalogue/config.md that had no sim proof and that the state line or a render can prove.
# One scenario, qa_config_rows; asserts only (a snap alone proves nothing). The sound document's test bank makes instrument 1 a SINE, so a pad's board note is heard at its pitch (m<note>).
document sound
audio on

# QA-CFG-219: F7 on the SD and INFO pages does nothing and says nothing (no F7 label there, no "NOTHING TO APPLY"). The overlay would be NOTIFY while a strip message shows.
tap config
tap f3
assert state screen=CONFIG page=2 overlay=NONE
tap f7
assert state screen=CONFIG page=2 overlay=NONE
tap f4
assert state screen=CONFIG page=3 overlay=NONE
tap f7
assert state screen=CONFIG page=3 overlay=NONE

# QA-CFG-242: Left on a USB row changes the pending value (starred, nothing written); the first F7 arms APPLY (the reconnect question on the strip, the key row's prompt),
# the second F7 writes it; a page change drops the pending value, so the next F7 is a no-op again.
tap f5
assert state screen=CONFIG page=4 overlay=NONE
tap left
assert state screen=CONFIG page=4 overlay=NONE
tap f7
assert state overlay=NOTIFY keys=F7:APPLY? msg=~RECONNECT
tap f7
assert state overlay=NOTIFY msg=~RECONNECTING
wait 5000
tap f1
tap f5
tap f7
assert state screen=CONFIG page=4 overlay=NOTIFY msg=NOTHING_TO_APPLY
wait 2500

# QA-CFG-230: a pad on Config (any page) auditions the current instrument at the pad board's note until the release; no setting changes, the instrument stays.
assert state screen=CONFIG instrument=1 overlay=NONE
render cfgr230 600
pad 24 down
wait 540
pad 24 up
wait 100
assert audio cfgr230 in=5-500 sound
assert audio cfgr230 in=560-600 silent
assert state screen=CONFIG instrument=1 overlay=NONE
render cfgr230off 400
wait 500
assert audio cfgr230off silent
assert state screen=CONFIG instrument=1

# QA-CFG-240: Left / Right change the value under the cursor, one detent, and the pad board follows at once: PAD OFFSET 12 -> 11; the asserts read the note the pad writes (48 after the detent).
tap f6
assert state screen=CONFIG page=5
tap down 7
tap pattern
action view.mode 0
action cursor.track 0
action cursor.home
assert state row=0
pad 24
assert model pat.1.0.0.note=49
tap config
tap f5
tap f6
assert state screen=CONFIG page=5
tap down 7
tap left
assert state screen=CONFIG page=5
tap pattern
action view.mode 0
action cursor.home
assert state row=1
pad 24
assert model pat.1.0.1.note=48
